STN32CubeIDE


Быстрое создание CMSIS проекта:
  • Запускаем STN32CubeIDE/
  • New/STM32Project
  • Ставим свой контроллер
  • Генерация кубом нам не нужна,поэтому Empty и Finish.
  • В папку INC кидаем наши CORE и CMSIS .h файлы.
  • Пишем main.c. Там ставим #include "stm32f4xx.h"
  • В файле stm32f4xx.h раскомментируем #define stm32f407xx и сохраняем его!
  • При сборке утверждаем ST-LINK. Всё!

/* Тест */
#include "stm32f4xx.h"

// Задержка в виде функции
void delay(void) {
  for(int i = 0; i < 50000*4; i++);
}

int main(void){
    // Настраиваем порты
    RCC->AHB1ENR |= RCC_AHB1ENR_GPIODEN;
    GPIOD->MODER |= GPIO_MODER_MODER12_0 |
         GPIO_MODER_MODER13_0  | GPIO_MODER_MODER14_0 |
         GPIO_MODER_MODER15_0;

   // Настраиваем тактирование от кварца
 RCC->CR |= RCC_CR_CSSON | RCC_CR_HSEON;// Включаем HSE и CSS
 RCC->CFGR |= RCC_CFGR_SW_HSE;       // Переключаемся на него

// Собственно работа программы в безконечном цикле
for(;;){
          GPIOD->BSRR |= GPIO_BSRR_BS_12;   delay();
          GPIOD->BSRR |= GPIO_BSRR_BS_13;   delay();
          GPIOD->BSRR |= GPIO_BSRR_BS_14;   delay();
          GPIOD->BSRR |= GPIO_BSRR_BS_15;   delay();
          // Гасим всё сразу
          GPIOD-> ODR = 0x0;                delay();
        }
}
